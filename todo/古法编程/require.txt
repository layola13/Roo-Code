## 核心功能设计

### 1. 增强型错误诊断系统
- **传统红线保留**：保持VSCode默认的错误红线提示，这是程序员熟悉的交互方式 
- **悬停增强**：鼠标悬停时显示简洁的错误类型和位置，不显示完整解决方案，避免干扰思考 
- **侧边栏深度分析**：点击错误后，侧边栏显示：
  - 错误原因的分层解析（从语法到逻辑）
  - 相关代码上下文关联
  - 历史相似错误案例（匿名化处理）
  - **关键创新**：显示"思考引导"而不是直接答案，比如"检查这个变量的作用域"、"考虑这个函数的返回类型是否匹配"

### 2. "古法"辅助功能

#### 代码脉络图
- 在侧边栏生成当前文件的**代码结构脉络图**，用树状结构展示函数、变量、依赖关系
- 不提供自动补全，只帮助程序员理清思路
- 支持手动画重点标记（类似纸质笔记）

#### 历史对比工具
- 保存代码修改前后的差异，但不自动应用
- 程序员可以手动对比不同版本，培养对代码演进的理解 

#### 上下文感知文档
- 当鼠标悬停在API或函数上时，在侧边栏显示**精简版文档**，只包含核心参数和返回值
- 避免信息过载，让程序员自己探索更多细节 

### 3. 创新交互设计

#### "沉思模式"
- 按下特定快捷键（如Ctrl+Shift+M）进入沉思模式
- 隐藏所有AI提示，只保留基础语法高亮
- 只在代码真正出错时才显示最基础的错误提示 

#### 错误分级系统
- **语法错误**：红线，强制提示（必须处理）
- **逻辑警告**：黄色虚线，需要主动点击才显示详情
- **风格建议**：灰色点线，完全可忽略
- 这种分级让程序员决定何时接受帮助 

#### "古籍"知识库
- 内置经典编程模式和反模式案例
- 当检测到特定代码模式时，侧边栏显示相关经典案例
- 不直接建议修改，而是展示"古人是如何处理类似问题的"

interface ProblemWithTimer {
    id: string;
    severity: 'error' | 'warning' | 'info';

    timeLimit: number; // 秒

// 根据用户级别和问题难度动态计算时间限制
function calculateTimeLimit(severity: string, difficulty: number, userLevel: string): number {
    const baseTimes = {
        'beginner': { 'error': 300, 'warning': 180, 'info': 60 }, // 新手：5分钟、3分钟、1分钟
        'professional': { 'error': 180, 'warning': 120, 'info': 30 }, // 专业：3分钟、2分钟、30秒
        'expert': { 'error': 120, 'warning': 60, 'info': 15 }, // 专家：2分钟、1分钟、15秒
        'professor': { 'error': 60, 'warning': 30, 'info': 5 } // 教授：1分钟、30秒、5秒
    };
    
    return baseTimes[userLevel][severity] * (1 + difficulty * 0.2);
}

## 🎯 3. 评分系统（核心创新）

```typescript
class ScoringEngine {
    private baseScores = {
        handWrittenFix: (difficulty: number, userLevel: string) => {
            const multipliers = {
                'beginner': 1.5,
                'professional': 1.2,
                'expert': 1.0,
                'professor': 0.8
            };
            return 1.0 * difficulty * multipliers[userLevel];
        },
        aiGeneratedFix: 0.01,
        timeBonus: (remainingTime: number, timeLimit: number) => {
            return remainingTime / timeLimit * 0.5; // 最多+0.5分时间奖励
        },
        timePenalty: (overTime: number, timeLimit: number) => {
            return -Math.min(overTime / timeLimit * 0.5, 0.5); // 最多-0.5分惩罚
        }
    };

    calculateScore(fixType: 'handwritten' | 'ai-generated', 
                  problem: ProblemWithTimer, 
                  userConfig: UserConfig): number {
        let score = 0;
        
        // 基础分
        if (fixType === 'handwritten') {
            score = this.baseScores.handWrittenFix(problem.difficulty, userConfig.skillLevel);
        } else {
            score = this.baseScores.aiGeneratedFix;
        }
        
        // 时间奖励/惩罚
        const elapsed = Date.now() - problem.startTime.getTime();
        if (elapsed <= problem.timeLimit * 1000) {
            score += this.baseScores.timeBonus(
                (problem.timeLimit * 1000 - elapsed) / 1000, 
                problem.timeLimit
            );
        } else {
            score += this.baseScores.timePenalty(
                (elapsed - problem.timeLimit * 1000) / 1000,
                problem.timeLimit
            );
        }
        
        return score;
    }
}
```

interface CommitQualityScore {
    formatScore: number;      // 提交格式规范性 (0-10)
    messageQuality: number;   // 信息详细度和质量 (0-10)
    codeMatchScore: number;   // 代码变更与提交信息匹配度 (0-10)
    granularityScore: number; // 提交粒度合理性 (0-10)
    overallScore: number;     // 综合评分 (0-100)
}

function analyzeCommitQuality(commitMessage: string, changedFiles: string[], diffStats: any): CommitQualityScore {
    // 1. 格式分析 (conventional commits)
    const formatScore = analyzeCommitFormat(commitMessage);
    
    // 2. 信息质量分析
    const messageQuality = analyzeMessageQuality(commitMessage);
    
    // 3. 代码匹配度分析
    const codeMatchScore = analyzeCodeMessageMatch(commitMessage, changedFiles);
    
    // 4. 粒度分析
    const granularityScore = analyzeCommitGranularity(diffStats);
    
    return {
        formatScore,
        messageQuality,
        codeMatchScore,
        granularityScore,
        overallScore: (formatScore + messageQuality + codeMatchScore + granularityScore) * 2.5
    };
}
```

```typescript
interface CommitTimeChallenge {
    type: 'bug-fix' | 'feature' | 'refactor' | 'hotfix';
    difficulty: number;
    timeLimit: number; // 分钟
    qualityThreshold: number; // 质量阈值 (0-100)
    rewardMultiplier: number;
}

function createCommitChallenge(commitType: string, legacyAnalysis: LegacyCodeAnalysis): CommitTimeChallenge {
    const baseChallenges = {
        'bug-fix': { timeLimit: 30, qualityThreshold: 85, rewardMultiplier: 1.2 },
        'feature': { timeLimit: 120, qualityThreshold: 90, rewardMultiplier: 1.5 },
        'refactor': { timeLimit: 60, qualityThreshold: 95, rewardMultiplier: 2.0 },
        'hotfix': { timeLimit: 15, qualityThreshold: 80, rewardMultiplier: 1.8 }
    };
    
    // 根据代码库成熟度调整难度
    const challenge = baseChallenges[commitType];
    if (legacyAnalysis.codebaseMaturity < 40) {
        // 高技术债务，增加时间，降低质量要求
        challenge.timeLimit *= 1.5;
        challenge.qualityThreshold *= 0.9;
    }
    
    return {
        ...challenge,
        difficulty: calculateCommitDifficulty(commitType, legacyAnalysis),
        type: commitType
    };
}
```

function calculateGitCollaborationScore(commitQuality: CommitQualityScore, 
                                      timeChallenge: CommitTimeChallenge, 
                                      userLevel: string): number {
    // 基础分：提交质量
    const baseScore = commitQuality.overallScore / 10;
    
    // 时间奖励/惩罚
    let timeBonus = 0;
    if (commitQuality.timeTaken <= timeChallenge.timeLimit * 0.5) {
        timeBonus = 2.0; // 50%时间内完成
    } else if (commitQuality.timeTaken <= timeChallenge.timeLimit) {
        timeBonus = 1.0; // 按时完成
    } else {
        timeBonus = -0.5 * Math.min((commitQuality.timeTaken - timeChallenge.timeLimit) / 10, 2); // 超时惩罚
    }
    
    // 质量奖励
    let qualityBonus = 0;
    if (commitQuality.overallScore >= timeChallenge.qualityThreshold + 10) {
        qualityBonus = 3.0; // 超高质量
    } else if (commitQuality.overallScore >= timeChallenge.qualityThreshold) {
        qualityBonus = 1.5; // 达到标准
    }
    
    // 团队协作加成
    const teamBonus = calculateTeamCollaborationBonus();
    
    return (baseScore + timeBonus + qualityBonus + teamBonus) * timeChallenge.rewardMultiplier;
}

## 🎯 一、核心设计哲学

### 1.1 核心理念：**辅助而非替代**
- **传统AI工具**：生成代码 → 程序员审核 → 快速完成
- **古法编程**：发现问题 → 引导思考 → 程序员手写 → 能力提升
- **关键差异**：不是"帮你写代码"，而是"帮你成为更好的程序员"

### 1.2 设计原则
- **最小干预原则**：只在必要时提供帮助，避免过度干扰
- **渐进式提示**：从最简提示开始，程序员需要时才展开详情
- **成就感驱动**：通过游戏化设计，让手写代码变得有成就感
- **真实能力评估**：分数反映真实编程能力，而非AI使用频率

### 4.2 提交质量评估逻辑
**多维度评分**：
1. **格式规范性** (25%)：是否符合团队提交规范
2. **信息完整性** (25%)：是否描述清楚变更原因和影响
3. **代码匹配度** (30%)：提交信息与代码变更是否一致
4. **粒度合理性** (20%)：是否原子化提交，避免过大变更

**智能反馈**：
- 90+分：显示"完美提交！+5分，解锁成就进度+10%"
- 70-89分：显示"良好提交，建议改进：... +3分"
- 50-69分：显示"需要改进：... +1分"
- <50分：显示"重新考虑提交策略，建议拆分或补充说明 0分"

## 💎 七、核心设计价值总结

### 7.1 UI设计原则
- **最小干扰**：默认折叠详细信息，只显示最关键内容
- **渐进披露**：点击展开更多细节，保持界面清爽
- **情感化设计**：使用表情符号、颜色、动画增强情感连接
- **上下文感知**：根据用户水平、项目类型自动调整界面复杂度
- **行动导向**：每个界面都有明确的下一步行动建议

### 7.2 提示词设计哲学
- **新手友好**：详细解释、鼓励性语言、降低压力
- **专家挑战**：哲学思考、深入分析、适度压力
- **古法精神**：强调手写价值、能力提升、长期收益
- **团队协作**：突出贡献、集体成就、互相学习
- **真实反馈**：不粉饰问题，但提供改进路径

### 7.3 差异化优势
```
传统AI工具 UI vs 古法编程 UI：
┌─────────────────────────────────────────────────────────────┐
│ 传统AI工具                  │ 古法编程                          │
├─────────────────────────────┼───────────────────────────────────┤
│ 代码补全面板居中            │ 问题分析面板居中                  │
│ "生成代码"按钮最大          │ "最简提示"按钮最大                │
│ 强调速度和效率              │ 强调质量和能力提升                │
│ 个人成就为主                │ 个人+团队成就并重                 │
│ 简单的分数显示              │ 多维度能力雷达图                  │
│ 无时间压力                  │ 智能时间激励                      │
│ 代码生成历史                │ 手写修复历史                      │
└─────────────────────────────┴───────────────────────────────────┘
```

这个UI设计不仅仅是界面，而是一个**完整的编程能力培养系统**。它通过精心设计的提示词、交互元素和数据可视化，让程序员在享受"你打败了99%的程序员"成就感的同时，真正回归手写代码的本质，提升真实编程能力。

                          </div>
                          <div className="mt-3 bg-gray-50 p-3 rounded-lg border border-dashed border-gray-300">
                            <div className="text-sm text-gray-600">
                              <strong className="text-gray-800">📊 修复预估：</strong>
                              <span className="ml-2 text-green-600 font-medium">手写+{problem.estimatedReward}分</span> | 
                              <span className="ml-2 text-blue-600 font-medium">AI生成+{problem.aiReward}分</span> | 
                              <span className="ml-2 text-red-600 font-medium">超时-{problem.overtimePenalty}分</span>
                            </div>
                          </div>
                        </div>
                      </div>
                    </div>
                    <div className="flex flex-col md:w-auto w-full gap-2 mt-3 md:mt-0">
                      <button className="bg-indigo-600 hover:bg-indigo-700 text-white font-medium py-2 px-4 rounded-lg transition-colors duration-200 flex items-center justify-center gap-2 shadow-md">
                        💡 最简提示
                      </button>
                      <button className={`bg-gray-200 hover:bg-gray-300 text-gray-800 font-medium py-2 px-4 rounded-lg transition-colors duration-200 flex items-center justify-center gap-2 ${problem.timeLeft !== null ? 'block' : 'hidden'}`}>
                        ⏱️ 延长时间
                      </button>
                    </div>
                  </div>
                </div>
              ))}
            </div>
          </div>

          {/* Warnings */}
          <div className="bg-white rounded-xl shadow-sm border border-yellow-100 overflow-hidden">
            <div className="bg-yellow-50 px-6 py-4 border-b border-yellow-100 flex justify-between items-center">
              <div className="flex items-center gap-2">
                <div className="text-yellow-500 text-xl font-bold">🟡</div>
                <h2 className="text-lg font-bold text-gray-800">警告 (3)</h2>
              </div>
              <div className="px-3 py-1 rounded-full font-medium text-sm bg-green-100 text-green-800">
                📈 今日进步：+{userProfile.todayScore}分
              </div>
            </div>
            <div className="divide-y divide-gray-100">
              {problems.filter(p => p.type === 'warning').map((problem) => (
                <div key={problem.id} className="p-4 hover:bg-yellow-50 transition-colors duration-200">
                  <div className="flex items-start gap-3">
                    <div className="text-xl mt-1">{getSeverityIcon(problem.type)}</div>
                    <div>
                      <div className="font-medium text-gray-900">
                        [{problem.id}] {problem.title}
                      </div>
                      <div className="text-gray-600 mt-1 text-sm">
                        {problem.location}
                      </div>
                    </div>
                  </div>
                </div>
              ))}
            </div>
          </div>

          {/* Improvements */}
          <div className="bg-white rounded-xl shadow-sm border border-blue-100 overflow-hidden">
            <div className="bg-blue-50 px-6 py-4 border-b border-blue-100 flex justify-between items-center">
              <div className="flex items-center gap-2">
                <div className="text-blue-500 text-xl font-bold">🔵</div>
                <h2 className="text-lg font-bold text-gray-800">建议 (2)</h2>
              </div>
              <div className="px-3 py-1 rounded-full font-medium text-sm bg-purple-100 text-purple-800">
                🎯 今日目标：+{userProfile.todayGoal}分
              </div>
            </div>
            <div className="divide-y divide-gray-100">
              {problems.filter(p => p.type === 'improvement' || p.type === 'info').map((problem) => (
                <div key={problem.id} className="p-4 hover:bg-blue-50 transition-colors duration-200">
                  <div className="flex items-start gap-3">
                    <div className="text-xl mt-1">{getSeverityIcon(problem.type)}</div>
                    <div>
                      <div className="font-medium text-gray-900">
                        [{problem.id}] {problem.title}
                      </div>
                      <div className="text-gray-600 mt-1 text-sm">
                        {problem.location}
                      </div>
                    </div>
                  </div>
                </div>
              ))}
            </div>
          </div>
        </div>

        {/* Statistics and Actions */}
        <div className="grid grid-cols-1 md:grid-cols-4 gap-4 mb-8">
          <div className="bg-white rounded-xl shadow-sm p-4 border border-green-100">
            <div className="text-sm text-gray-600 mb-1">📊 实时统计</div>
            <div className="text-2xl font-bold text-green-600">{userProfile.handwriteRate}%</div>
            <div className="text-sm text-gray-700 mt-1">手写率</div>
            <div className="text-xs text-gray-500 mt-0.5">AI依赖: {userProfile.aiDependence}%</div>
          </div>
          
          <div className="bg-white rounded-xl shadow-sm p-4 border border-blue-100">
            <div className="text-sm text-gray-600 mb-1">⏱️ 今日效率</div>
            <div className="text-2xl font-bold text-blue-600">{userProfile.avgFixTime}s</div>
            <div className="text-sm text-gray-700 mt-1">平均修复时间</div>
            <div className="text-xs text-gray-500 mt-0.5">超时率: {userProfile.overtimeRate}%</div>
          </div>
          
          <div className="bg-white rounded-xl shadow-sm p-4 border border-amber-100">
            <div className="text-sm text-gray-600 mb-1">🏅 成就进度</div>
            <div className="text-2xl font-bold text-amber-600">🥇 95%</div>
            <div className="text-sm text-gray-700 mt-1">手写大师</div>
            <div className="text-xs text-gray-500 mt-0.5">下一成就: 闪电手(80%)</div>
          </div>
          
          <div className="bg-white rounded-xl shadow-sm p-4 border border-purple-100">
            <div className="text-sm text-gray-600 mb-1">👥 团队贡献</div>
            <div className="text-2xl font-bold text-purple-600">📝 {userProfile.commitsToday}/{userProfile.commitsGoal}</div>
            <div className="text-sm text-gray-700 mt-1">今日提交</div>
            <div className="text-xs text-gray-500 mt-0.5">团队排名: #{userProfile.teamRank}/{userProfile.teamSize}</div>
          </div>
        </div>

        {/* Action Buttons */}
        <div className="flex flex-wrap justify-center gap-3 mb-12">
          <button className="bg-indigo-600 hover:bg-indigo-700 text-white font-medium py-3 px-6 rounded-xl transition-all duration-200 shadow-lg flex items-center gap-2 text-lg">
            💡 最简提示
          </button>
          <button className="bg-amber-500 hover:bg-amber-600 text-white font-medium py-3 px-6 rounded-xl transition-all duration-200 shadow-lg flex items-center gap-2 text-lg">
            ⏱️ 延长时间
          </button>
          <button className="bg-blue-500 hover:bg-blue-600 text-white font-medium py-3 px-6 rounded-xl transition-all duration-200 shadow-lg flex items-center gap-2 text-lg">
            🔍 详细分析
          </button>
          <button className="bg-gray-300 hover:bg-gray-400 text-gray-800 font-medium py-3 px-6 rounded-xl transition-all duration-200 shadow flex items-center gap-2 text-lg">
            🤖 AI生成(不推荐)
          </button>
          <button className="bg-emerald-500 hover:bg-emerald-600 text-white font-medium py-3 px-6 rounded-xl transition-all duration-200 shadow-lg flex items-center gap-2 text-lg">
            📊 全部统计
          </button>
        </div>

        {/* Achievements Section */}
        <div className="bg-white rounded-xl shadow-lg p-6 mb-12 border border-gray-200">
          <div className="flex justify-between items-center mb-5">
            <h2 className="text-xl font-bold text-gray-800 flex items-center gap-2">
              🏆 您的成就殿堂 <span className="text-sm font-normal text-gray-600">(23/50)</span>
            </h2>
            <button className="text-indigo-600 hover:text-indigo-800 font-medium flex items-center gap-1">
              查看全部 <span className="ml-1">→</span>
            </button>
          </div>
          
          <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
            {achievements.slice(0, 3).map((achievement) => (
              <div key={achievement.id} className={`border-2 rounded-xl p-4 transition-all duration-200 ${getAchievementColor(achievement)}`}>
                <div className="flex justify-between items-start mb-2">
                  <div className={`text-2xl ${achievement.unlocked ? 'text-green-500' : 'text-gray-400'}`}>
                    {achievement.unlocked ? '🏆' : '🎖️'}
                  </div>
                  <span className={`text-xs font-medium px-2 py-1 rounded-full ${
                    achievement.type === 'personal' ? 'bg-blue-100 text-blue-800' : 'bg-purple-100 text-purple-800'
                  }`}>
                    {achievement.type === 'personal' ? '个人成就' : '团队成就'}
                  </span>
                </div>
                <h3 className="text-lg font-bold text-gray-900 mb-1">{achievement.name}</h3>
                <p className="text-gray-600 text-sm mb-3">{achievement.description}</p>
                <div className="flex justify-between items-center">
                  <div className="text-sm font-medium">
                    {getAchievementStatus(achievement)}
                  </div>
                  <div className={`font-bold ${
                    achievement.unlocked ? 'text-green-600' : 'text-indigo-600'
                  }`}>
                    +{achievement.reward}分
                  </div>
                </div>
              </div>
            ))}
          </div>
        </div>

        {/* Footer */}
        <div className="text-center text-gray-500 text-sm py-4 border-t border-gray-200">
          <p>🏮 古法编程 v1.0 - 代码之道，贵在手写 | 今日已帮助 {1523} 位程序员提升手写能力</p>
          <p className="mt-1">💪 真正的程序员相信自己的双手，而非AI的生成</p>
        </div>
      </div>
    </div>
  );
};

export default App;
